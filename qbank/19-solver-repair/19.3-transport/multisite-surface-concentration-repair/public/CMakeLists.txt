cmake_minimum_required(VERSION 3.18)
project(cantera_flow_repair LANGUAGES CXX)
file(GLOB_RECURSE upstream_objects /opt/cantera-source/build/src/*.os)
list(FILTER upstream_objects EXCLUDE REGEX "/(FlowReactor|SurfPhase|canteraStatic)\.os$")
set_source_files_properties(${upstream_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_library(cantera_repair SHARED /opt/cantera-source/src/zeroD/FlowReactor.cpp /opt/cantera-source/src/thermo/SurfPhase.cpp ${upstream_objects})
target_compile_features(cantera_repair PRIVATE cxx_std_17)
target_compile_options(cantera_repair PRIVATE -O3)
target_include_directories(cantera_repair PRIVATE /opt/cantera-source/include /usr/include/eigen3)
target_link_libraries(cantera_repair PRIVATE fmt yaml-cpp openblas gomp sundials_cvodes sundials_idas sundials_nvecserial sundials_core sundials_sunlinsoldense sundials_sunlinsolband dl)
set_target_properties(cantera_repair PROPERTIES OUTPUT_NAME cantera_shared VERSION 3.1.0 SOVERSION 3 LIBRARY_OUTPUT_DIRECTORY /opt/cantera1996-build/lib)
add_executable(flow_driver /opt/cantera1996-driver/driver.cpp)
target_compile_features(flow_driver PRIVATE cxx_std_17)
target_include_directories(flow_driver PRIVATE /opt/cantera-source/include /usr/include/eigen3)
target_link_libraries(flow_driver PRIVATE cantera_repair fmt)
set_target_properties(flow_driver PROPERTIES OUTPUT_NAME flow-driver RUNTIME_OUTPUT_DIRECTORY /opt/cantera1996-build/bin)
add_custom_target(repair_sources DEPENDS cantera_repair flow_driver)
