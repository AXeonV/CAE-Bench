cmake_minimum_required(VERSION 3.18)
project(cantera_header_repair LANGUAGES CXX)
# All unaffected objects were compiled from the bundled upstream tree.
file(GLOB_RECURSE upstream_objects /opt/cantera-source/build/src/*.os)
list(FILTER upstream_objects EXCLUDE REGEX "/(PlasmaPhase|ThermoFactory|ElectronCollisionPlasmaRate|ReactionRateFactory|canteraStatic)\\.os$")
set_source_files_properties(${upstream_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
add_library(cantera_repair SHARED /opt/cantera-source/src/kinetics/ElectronCollisionPlasmaRate.cpp /opt/cantera-source/src/kinetics/ReactionRateFactory.cpp /opt/cantera-source/src/thermo/PlasmaPhase.cpp /opt/cantera-source/src/thermo/ThermoFactory.cpp ${upstream_objects})
target_compile_features(cantera_repair PRIVATE cxx_std_17)
target_compile_options(cantera_repair PRIVATE -O3)
target_include_directories(cantera_repair PRIVATE /opt/cantera-source/include /usr/include/eigen3)
target_link_libraries(cantera_repair PRIVATE fmt yaml-cpp openblas gomp sundials_cvodes sundials_idas sundials_nvecserial sundials_core sundials_sunlinsoldense sundials_sunlinsolband dl)
set_target_properties(cantera_repair PROPERTIES OUTPUT_NAME cantera_shared VERSION 3.1.0 SOVERSION 3 LIBRARY_OUTPUT_DIRECTORY /opt/cantera-source/build/lib)
add_executable(plasma_driver /opt/repair-driver/driver.cpp)
target_compile_features(plasma_driver PRIVATE cxx_std_17)
target_include_directories(plasma_driver PRIVATE /opt/cantera-source/include /usr/include/eigen3)
target_link_libraries(plasma_driver PRIVATE cantera_repair)
set_target_properties(plasma_driver PROPERTIES OUTPUT_NAME plasma-driver RUNTIME_OUTPUT_DIRECTORY /opt/cantera-source/build/bin)
add_custom_target(repair_sources DEPENDS cantera_repair plasma_driver)
