cmake_minimum_required(VERSION 3.18)
project(precice_projection_repair LANGUAGES CXX)
file(GLOB_RECURSE upstream_objects /opt/precice-build/CMakeFiles/precice.dir/*.o)
list(FILTER upstream_objects EXCLUDE REGEX "/query/Index[.]cpp[.]o$")
set_source_files_properties(${upstream_objects} PROPERTIES EXTERNAL_OBJECT TRUE GENERATED TRUE)
set(SOURCE_ROOT /opt/precice-upstream/precice-c2a06a65631e15a6622f5ecffae194f82ba0bae3)
add_library(precice_repair SHARED ${SOURCE_ROOT}/src/query/Index.cpp ${upstream_objects})
find_package(Boost 1.71 REQUIRED COMPONENTS log_setup program_options system unit_test_framework log filesystem thread atomic chrono regex)
find_package(LibXml2 REQUIRED)
target_compile_features(precice_repair PRIVATE cxx_std_17)
target_compile_options(precice_repair PRIVATE -O1)
target_compile_definitions(precice_repair PRIVATE BOOST_ALL_DYN_LINK BOOST_ASIO_ENABLE_OLD_SERVICES BOOST_GEOMETRY_DISABLE_DEPRECATED_03_WARNING FMT_HEADER_ONLY=1 PRECICE_NO_MPI PRECICE_NO_PETSC PRECICE_NO_PYTHON _GNU_SOURCE)
target_include_directories(precice_repair PRIVATE ${SOURCE_ROOT}/src /opt/precice-build/src /usr/include/eigen3 ${LIBXML2_INCLUDE_DIR} ${SOURCE_ROOT}/thirdparty/json/include ${SOURCE_ROOT}/thirdparty/fmt/include)
target_link_libraries(precice_repair PRIVATE ${Boost_LIBRARIES} ${LIBXML2_LIBRARIES} dl)
set_target_properties(precice_repair PROPERTIES OUTPUT_NAME precice VERSION 2.5.0 SOVERSION 2 LIBRARY_OUTPUT_DIRECTORY /opt/precice-repair-build/lib)
add_executable(projection_driver /opt/precice-driver/driver.cpp)
target_compile_features(projection_driver PRIVATE cxx_std_17)
target_compile_options(projection_driver PRIVATE -O1)
target_compile_definitions(projection_driver PRIVATE BOOST_ALL_DYN_LINK FMT_HEADER_ONLY=1 PRECICE_NO_MPI PRECICE_NO_PETSC PRECICE_NO_PYTHON)
target_include_directories(projection_driver PRIVATE ${SOURCE_ROOT}/src /opt/precice-build/src /usr/include/eigen3 ${SOURCE_ROOT}/thirdparty/json/include ${SOURCE_ROOT}/thirdparty/fmt/include)
target_link_libraries(projection_driver PRIVATE precice_repair)
set_target_properties(projection_driver PROPERTIES OUTPUT_NAME projection-driver RUNTIME_OUTPUT_DIRECTORY /opt/precice-repair-build/bin)
add_custom_target(repair_sources DEPENDS precice_repair projection_driver)
