project(meep_cylindrical_repair LANGUAGES CXX)
add_custom_target(meep_upstream COMMAND /usr/bin/make -C /opt/meep-repair-build/upstream/src --jobs=4 VERBATIM)

add_custom_target(repair_sources DEPENDS cylindrical_driver)
